Try it

Type a number or use − / +. Limits: 0–10.

Motion

When − or + is tapped, the button gives a small squeeze, and the old number slides away and fades out while the new one slides in from the other side. Source: Component Motion Guidelines › Numeric Stepper ↗

Two things happen on every tap, starting together:

  • Button squeeze (334 ms): the tapped button shrinks to 90%, then springs back to 100% with a slight overshoot.
  • Number roll (200 ms): the digits travel only 3.75 px, so it reads as a soft "tick", not a slide. + moves them down (the new number drops in from above); − mirrors it and moves them up.

Rules

  • Values are in px (web) = pt (iOS) = dp (Android).
  • Only digits that change roll. Digits are compared position by position: 12 → 13 rolls just the last digit, 19 → 20 rolls both, 9 → 10 rolls both (the "0" is new).
  • Old and new digits overlap at the same spot; nothing clips them.
  • Typing a number does not animate, only − / + do. Going from an empty field to the first number doesn't animate either (the label floats instead).
  • The button scales from its center. Only the tapped button animates; a disabled one (at the min / max limit) doesn't.
  • Reduce Motion / Remove animations on: no squeeze, swap the number instantly.

Each digit position gets its own ZStack keyed by the digit, so only digits that change roll, and the old and new digit overlap during the transition. Pass the direction of the last tap.

import SwiftUI

extension Animation {
    // Component Motion Guidelines › Numeric Stepper
    static let stepperMove = Animation.timingCurve(0.33, 0, 0.25, 1, duration: 0.167)
    static let stepperFade = Animation.timingCurve(0.33, 0, 0.25, 1, duration: 0.083)
}

extension AnyTransition {
    /// direction: 1 after "+", -1 after "−"
    static func stepperRoll(_ direction: CGFloat) -> AnyTransition {
        let shift: CGFloat = 3.75
        return .asymmetric(
            insertion: AnyTransition.offset(y: -shift * direction).animation(.stepperMove.delay(0.033))
                .combined(with: AnyTransition.opacity.animation(.stepperFade.delay(0.033))),
            removal: AnyTransition.offset(y: shift * direction).animation(.stepperMove)
                .combined(with: AnyTransition.opacity.animation(.stepperFade))
        )
    }
}

struct StepperValue: View {
    let value: Int
    let direction: CGFloat
    @Environment(\.accessibilityReduceMotion) private var reduceMotion

    var body: some View {
        let digits = Array(String(value))
        let roll: AnyTransition = reduceMotion ? .identity : .stepperRoll(direction)
        // One slot per digit position: only a digit that changes gets a new id, so only it rolls
        // (12 → 13 rolls just the "3"; 19 → 20 rolls both).
        HStack(spacing: 0) {
            ForEach(digits.indices, id: \.self) { i in
                ZStack {
                    Text(String(digits[i]))
                        .id(digits[i])
                        .transition(roll)
                }
                .transition(roll)   // a new position (9 → 10) rolls in too
            }
        }
    }
}

// Button squeeze: 100% → 90% (167 ms) → 100% (167 ms, overshoot). iOS 17+.
struct StepperBump: ViewModifier {
    let trigger: Int   // bump it on every tap
    @Environment(\.accessibilityReduceMotion) private var reduceMotion

    func body(content: Content) -> some View {
        if reduceMotion { content } else {
            content.keyframeAnimator(initialValue: CGFloat(1), trigger: trigger) { view, scale in
                view.scaleEffect(scale)
            } keyframes: { _ in
                KeyframeTrack(\.self) {
                    LinearKeyframe(0.9, duration: 0.167, timingCurve: .bezier(
                        startControlPoint: UnitPoint(x: 0.33, y: 0), endControlPoint: UnitPoint(x: 0.25, y: 1)))
                    LinearKeyframe(1.0, duration: 0.167, timingCurve: .bezier(
                        startControlPoint: UnitPoint(x: 0.44, y: 0), endControlPoint: UnitPoint(x: 0.30, y: 2.08)))
                }
            }
        }
    }
}

Each digit position has its own AnimatedContent, so only the digits that change animate. Pass the direction of the last tap. Note: a digit position that is added (9 → 10) appears without the roll in this version.

import androidx.compose.animation.*
import androidx.compose.animation.core.Animatable
import androidx.compose.animation.core.CubicBezierEasing
import androidx.compose.animation.core.tween
import androidx.compose.material3.IconButton
import androidx.compose.material3.Text
import androidx.compose.foundation.layout.Row
import androidx.compose.runtime.Composable
import androidx.compose.runtime.key
import androidx.compose.runtime.remember
import androidx.compose.runtime.rememberCoroutineScope
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.graphicsLayer
import androidx.compose.ui.platform.LocalDensity
import androidx.compose.ui.unit.dp
import kotlinx.coroutines.launch

// Component Motion Guidelines › Numeric Stepper
private val StepperEasing = CubicBezierEasing(0.33f, 0f, 0.25f, 1f)
private const val MOVE_MS = 167
private const val FADE_MS = 83
private const val DELAY_MS = 33

@Composable
fun StepperValue(value: Int, direction: Int, modifier: Modifier = Modifier) {   // direction: 1 after "+", -1 after "−"
    val shift = with(LocalDensity.current) { 3.75.dp.roundToPx() }
    // One AnimatedContent per digit position: only a digit that changes animates
    // (12 → 13 rolls just the "3"; 19 → 20 rolls both).
    Row(modifier) {
        value.toString().forEachIndexed { i, digit ->
            key(i) {
                AnimatedContent(
                    targetState = digit,
                    transitionSpec = {
                        (slideInVertically(tween(MOVE_MS, DELAY_MS, StepperEasing)) { -shift * direction } +
                            fadeIn(tween(FADE_MS, DELAY_MS, StepperEasing))) togetherWith
                            (slideOutVertically(tween(MOVE_MS, 0, StepperEasing)) { shift * direction } +
                                fadeOut(tween(FADE_MS, 0, StepperEasing))) using
                            SizeTransform(clip = false)
                    },
                    label = "stepper-digit-$i",
                ) { Text(it.toString()) }
            }
        }
    }
}

// Button squeeze: 100% → 90% (167 ms) → 100% (167 ms, overshoot)
private val ReleaseEasing = CubicBezierEasing(0.44f, 0f, 0.30f, 2.08f)

@Composable
fun StepperButton(onClick: () -> Unit, enabled: Boolean = true, content: @Composable () -> Unit) {
    val scale = remember { Animatable(1f) }
    val scope = rememberCoroutineScope()
    IconButton(
        onClick = {
            onClick()
            scope.launch {
                scale.animateTo(0.9f, tween(MOVE_MS, 0, StepperEasing))
                scale.animateTo(1f, tween(MOVE_MS, 0, ReleaseEasing))
            }
        },
        enabled = enabled,
        modifier = Modifier.graphicsLayer { scaleX = scale.value; scaleY = scale.value },
    ) { content() }
}

Built into bog-ui.js: any .bog-number-input animates when − / + is tapped, so prototypes need no extra code. Under prefers-reduced-motion: reduce the number swaps instantly. The values, for reference:

const EASE = "cubic-bezier(0.33, 0, 0.25, 1)";
const dir = 1;  // + = 1 (down), − = -1 (up)

oldEl.animate({ transform: ["translateY(0)", `translateY(${3.75 * dir}px)`] }, { duration: 167, easing: EASE, fill: "forwards" });
oldEl.animate({ opacity: [1, 0] },                                          { duration: 83,  easing: EASE, fill: "forwards" });
newEl.animate({ transform: [`translateY(${-3.75 * dir}px)`, "translateY(0)"] }, { duration: 167, delay: 33, easing: EASE, fill: "backwards" });
newEl.animate({ opacity: [0, 1] },                                          { duration: 83,  delay: 33, easing: EASE, fill: "backwards" });

// Button squeeze, on the tapped − / + button
button.animate([
  { transform: "scale(1)",   easing: EASE },

  { transform: "scale(1)" }
