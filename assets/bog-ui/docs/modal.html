Try it

Open each layout. Close with the X, a button, Escape or a tap on the dim background.

Motion

The dialog grows in from 60% while rising 70 px and fading in over a dimming background; on close it shrinks to 62% and fades out. Every layout (Standard, Minimal, Cover, Destructive, Single / Dual) uses the same motion. Source: Dialog - Open ↗ · Dialog - Close ↗

Open takes 500 ms; the dialog is fully visible after 117 ms and keeps settling upward. Close is quicker: the dialog is gone after 133 ms, the dim background after 183 ms, and the shrink finishes at 333 ms.

Rules

  • All tracks start together (no delays). Scale is from the dialog's center.
  • Values are in px (web) = pt (iOS) = dp (Android). The overlay at 100% is the Support / Overlay canvas color (#0F0F10 at 68%).
  • Close does not move the dialog. If it's closed while still opening, it finishes its rise during the close.
  • Reopening while it closes animates back from where it is.
  • Same motion for every trigger: X, a footer button, a tap on the overlay, Escape / back.
  • Reduce Motion / Remove animations on: show and hide instantly.

A modifier that draws the overlay and the dialog above your screen. Each property gets its own withAnimation because they have different durations. iOS 17+ (for the completion).

import SwiftUI

// Component Motion Guidelines › Dialog - Open / Close
struct BOGDialogPresenter<Dialog: View>: ViewModifier {
    @Binding var isPresented: Bool
    @ViewBuilder var dialog: () -> Dialog

    // Resting (closed) values = where the open animation starts
    @State private var scale: CGFloat = 0.6
    @State private var offsetY: CGFloat = 70
    @State private var opacity: Double = 0
    @State private var overlay: Double = 0
    @Environment(\.accessibilityReduceMotion) private var reduceMotion

    private let overlayColor = Color(red: 15 / 255, green: 15 / 255, blue: 16 / 255).opacity(0.68)

    func body(content: Content) -> some View {
        content.overlay {
            ZStack {
                overlayColor.opacity(overlay).ignoresSafeArea()
                    .onTapGesture { isPresented = false }
                dialog()
                    .scaleEffect(scale)
                    .offset(y: offsetY)
                    .opacity(opacity)
                    .padding(16)
            }
            .allowsHitTesting(isPresented)
            .accessibilityHidden(!isPresented)
        }
        .onChange(of: isPresented) { _, open in open ? present() : dismiss() }
    }

    private func present() {
        if reduceMotion { scale = 1; offsetY = 0; opacity = 1; overlay = 1; return }
        let ease = { (s: Double) in Animation.timingCurve(0.33, 0, 0.09, 1, duration: s) }
        withAnimation(ease(0.333)) { scale = 1 }
        withAnimation(ease(0.5))   { offsetY = 0 }
        withAnimation(ease(0.117)) { opacity = 1; overlay = 1 }
    }

    private func dismiss() {
        let reset = { scale = 0.6; offsetY = 70; opacity = 0; overlay = 0 }
        if reduceMotion { reset(); return }
        withAnimation(.timingCurve(0.33, 0, 0.06, 1, duration: 0.333)) { scale = 0.62; offsetY = 0 } completion: {
            if !isPresented { reset() }   // back to the open start values, without animation
        }
        withAnimation(.timingCurve(0.17, 0, 0.41, 1, duration: 0.133)) { opacity = 0 }
        withAnimation(.timingCurve(0.17, 0, 0.83, 1, duration: 0.183)) { overlay = 0 }
    }
}

extension View {
    func bogDialog<D: View>(isPresented: Binding<Bool>, @ViewBuilder dialog: @escaping () -> D) -> some View {
        modifier(BOGDialogPresenter(isPresented: isPresented, dialog: dialog))
    }
}

// Usage
// @State private var showDialog = false
// MyScreen()
//     .bogDialog(isPresented: $showDialog) { MyDialogCard(onClose: { showDialog = false }) }

A host composable that draws the overlay and the dialog above your screen and runs each property as its own Animatable (different durations and easings).

import androidx.compose.animation.core.Animatable
import androidx.compose.animation.core.CubicBezierEasing
import androidx.compose.animation.core.tween
import androidx.compose.foundation.clickable
import androidx.compose.foundation.gestures.detectTapGestures
import androidx.compose.foundation.interaction.MutableInteractionSource
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.padding
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.draw.drawBehind
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.graphics.graphicsLayer
import androidx.compose.ui.input.pointer.pointerInput
import androidx.compose.ui.unit.dp
import kotlinx.coroutines.coroutineScope
import kotlinx.coroutines.launch

// Component Motion Guidelines › Dialog - Open / Close
private val OpenEasing = CubicBezierEasing(0.33f, 0f, 0.09f, 1f)
private val CloseScaleEasing = CubicBezierEasing(0.33f, 0f, 0.06f, 1f)
private val CloseFadeEasing = CubicBezierEasing(0.17f, 0f, 0.41f, 1f)
private val CloseOverlayEasing = CubicBezierEasing(0.17f, 0f, 0.83f, 1f)
private val OverlayColor = Color(0xAD0F0F10)   // Support / Overlay canvas (#0F0F10 @ 68%)

@Composable
fun BogDialogHost(visible: Boolean, onDismiss: () -> Unit, dialog: @Composable () -> Unit) {
    var shown by remember { mutableStateOf(visible) }
    // Resting (closed) values = where the open animation starts
    val scale = remember { Animatable(0.6f) }
    val offsetY = remember { Animatable(70f) }   // dp
    val alpha = remember { Animatable(0f) }
    val overlay = remember { Animatable(0f) }

    LaunchedEffect(visible) {
        if (visible) {
            shown = true
            coroutineScope {
                launch { scale.animateTo(1f, tween(333, easing = OpenEasing)) }
                launch { offsetY.animateTo(0f, tween(500, easing = OpenEasing)) }
                launch { alpha.animateTo(1f, tween(117, easing = OpenEasing)) }
                launch { overlay.animateTo(1f, tween(117, easing = OpenEasing)) }
            }
        } else if (shown) {
            coroutineScope {
                launch { scale.animateTo(0.62f, tween(333, easing = CloseScaleEasing)) }
                launch { offsetY.animateTo(0f, tween(333, easing = CloseScaleEasing)) }   // only moves if closed mid-open
                launch { alpha.animateTo(0f, tween(133, easing = CloseFadeEasing)) }
                launch { overlay.animateTo(0f, tween(183, easing = CloseOverlayEasing)) }
            }
            shown = false
            scale.snapTo(0.6f); offsetY.snapTo(70f)
        }
    }
    if (!shown) return

    Box(
        Modifier
            .fillMaxSize()
            .drawBehind { drawRect(OverlayColor, alpha = OverlayColor.alpha * overlay.value) }
            .clickable(remember { MutableInteractionSource() }, indication = null, onClick = onDismiss)
            .padding(16.dp),
        contentAlignment = Alignment.Center,
    ) {
        Box(
            Modifier
                .graphicsLayer {
                    scaleX = scale.value; scaleY = scale.value
                    translationY = offsetY.value * density
                    this.alpha = alpha.value
                }
                .pointerInput(Unit) { detectTapGestures { } },   // taps on the dialog don't dismiss it
        ) { dialog() }
    }
}
// Compose follows the system "Remove animations" setting (animator duration scale 0) automatically.

// Usage
// var showDialog by remember { mutableStateOf(false) }
// Box { MyScreen(); BogDialogHost(showDialog, { showDialog = false }) { MyDialogCard() } }

Built into modal.css + bog-ui.js: BOG.openModal() / BOG.closeModal() (and the data-modal-open / data-modal-close attributes) run it, for every layout. Under prefers-reduced-motion: reduce it shows and hides instantly. The CSS:

/* closed = start of open */
.bog-modal { background: transparent; }
.bog-modal__dialog { opacity: 0; scale: 0.6; translate: 0 70px; }

/* open */
.bog-modal[data-open] {
  background-color: var(--support-overlay-canvas);
  transition: background-color 117ms cubic-bezier(0.33, 0, 0.09, 1);
}
.bog-modal[data-open] .bog-modal__dialog {
  opacity: 1; scale: 1; translate: 0 0;
  transition: scale 333ms cubic-bezier(0.33, 0, 0.09, 1),
              translate 500ms cubic-bezier(0.33, 0, 0.09, 1),
              opacity 117ms cubic-bezier(0.33, 0, 0.09, 1);
}

/* closing (bog-ui.js sets data-closing, then hides the modal when the transitions end) */
.bog-modal[data-closing] { transition: background-color 183ms cubic-bezier(0.17, 0, 0.83, 1); }
.bog-modal[data-closing] .bog-modal__dialog {
  scale: 0.62; translate: 0 0;
  transition: scale 333ms cubic-bezier(0.33, 0, 0.06, 1),
              translate 333ms cubic-bezier(0.33, 0, 0.06, 1),
              opacity 133ms cubic-bezier(0.17, 0, 0.41, 1);
}
